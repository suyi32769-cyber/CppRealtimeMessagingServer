cmake_minimum_required(VERSION 3.16)
project(CppRealtimeMessagingServer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(USOCKETS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/uWebSockets/uSockets)
file(GLOB USOCKETS_SOURCES CONFIGURE_DEPENDS ${USOCKETS}/src/*.c ${USOCKETS}/src/eventing/*.c ${USOCKETS}/src/crypto/*.c ${USOCKETS}/src/io_uring/*.c)
add_library(usockets STATIC ${USOCKETS_SOURCES})
target_include_directories(usockets PUBLIC ${USOCKETS}/src)
target_compile_definitions(usockets PUBLIC LIBUS_NO_SSL)
find_package(ZLIB REQUIRED)
add_executable(realtime_server src/main.cpp)
target_include_directories(realtime_server PRIVATE third_party/uWebSockets/src third_party/json/single_include)
target_link_libraries(realtime_server PRIVATE usockets ZLIB::ZLIB pthread)
set(SQLITECPP_INTERNAL_SQLITE OFF CACHE BOOL "Use system SQLite" FORCE)
set(SQLITECPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/SQLiteCpp EXCLUDE_FROM_ALL)
add_library(message_repository src/MessageRepository.cpp)
target_include_directories(message_repository PUBLIC src)
target_link_libraries(message_repository PUBLIC SQLiteCpp)
target_link_libraries(realtime_server PRIVATE message_repository)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build bundled dependencies statically" FORCE)
set(DISABLE_TESTS ON CACHE BOOL "Disable bundled hiredis tests" FORCE)
add_subdirectory(third_party/hiredis EXCLUDE_FROM_ALL)
target_sources(realtime_server PRIVATE src/RedisBridge.cpp src/ConnectionManager.cpp src/MessageDispatcher.cpp)
target_link_libraries(realtime_server PRIVATE hiredis::hiredis)
add_executable(dependency_smoke tests/dependency_smoke.cpp)
target_include_directories(dependency_smoke PRIVATE third_party/json/single_include)
target_link_libraries(dependency_smoke PRIVATE SQLiteCpp hiredis::hiredis)
include(CTest)
add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
add_executable(repository_test tests/repository_test.cpp)
target_link_libraries(repository_test PRIVATE message_repository GTest::gtest_main)
add_executable(protocol_test tests/protocol_test.cpp)
target_include_directories(protocol_test PRIVATE src third_party/json/single_include)
target_link_libraries(protocol_test PRIVATE GTest::gtest_main)
add_executable(recent_events_test tests/recent_events_test.cpp)
target_include_directories(recent_events_test PRIVATE src)
target_link_libraries(recent_events_test PRIVATE GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(repository_test)
gtest_discover_tests(protocol_test)
gtest_discover_tests(recent_events_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME phase1_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_phase1.py)
add_test(NAME phase2_rooms COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/phase2.py)
add_test(NAME phase4_multi_instance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/phase4.py)
add_test(NAME room_membership COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/room_membership.py)
add_test(NAME robustness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/robustness.py)
add_test(NAME launcher COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/launcher.py)
add_test(NAME heartbeat COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_limits.py heartbeat)
add_test(NAME backpressure COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_limits.py backpressure)
set_tests_properties(phase1_smoke phase2_rooms phase4_multi_instance room_membership robustness launcher PROPERTIES RUN_SERIAL TRUE TIMEOUT 45)
set_tests_properties(heartbeat backpressure PROPERTIES RUN_SERIAL TRUE TIMEOUT 80)
